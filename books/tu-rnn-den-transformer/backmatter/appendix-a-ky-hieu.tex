\section{Hai chỗ dễ đọc nhầm ở Pascanu 2013}
\label{sec:app-a-pascanu}

\begin{itemize}
  \item \textbf{Chữ \(x\) đổi vai.} Biến trạng thái của bài là
    \emph{tiền kích hoạt} của sách, không phải trạng thái ẩn, và input của bài
    là \(\vect{u}_t\). Đặt \(\vect{a}_t\) là đối số của \(\sigma\) trong
    dạng quen thuộc \(\vect{h}_t = \sigma(\mat{W}_{hh}\vect{h}_{t-1} + \mat{W}_{xh}\vect{x}_t + \vect{b}_h)\)
    thì được \(\vect{a}_t = \mat{W}_{hh}\sigma(\vect{a}_{t-1}) + \mat{W}_{xh}\vect{x}_t + \vect{b}_h\),
    trùng khít phương trình của họ. Đây là phép đổi tên, không phải xấp xỉ.
  \item \textbf{\(\lambda_1\) không ánh xạ sang một ký hiệu duy nhất.} Bài
    phát biểu điều kiện đủ qua trị riêng lớn nhất, nhưng chứng minh của bài chặn
    giá trị kỳ dị lớn nhất, và với ma trận không chuẩn tắc đó là hai số khác
    nhau. Sách tách \(\rho(\mat{W}_{hh})\) khỏi \(\norm{\mat{W}_{hh}}\) và
    không bao giờ dùng lẫn; xem mục~\ref{sec:ch03-khe-ho}.
\end{itemize}

\section{Ba chỗ dễ đọc nhầm ở LSTM 1997}
\label{sec:app-a-lstm}

\begin{itemize}
  \item \textbf{Chữ \(h\) đổi vai.} Bài dùng \(h\) cho hàm nén output của ô
    nhớ, khoảng giá trị \([-1,1]\); sách dùng \(\vect{h}_t\) cho trạng thái ẩn.
    Sách viết hai hàm nén là \(g_{\mathrm{in}}\) và \(g_{\mathrm{out}}\), tương
    ứng với \(g\) và \(h\) của bài. Đọc bài với ký hiệu của sách trong đầu là
    cách nhanh nhất để lẫn hàm với trạng thái.
  \item \textbf{Bảng 10 mâu thuẫn với phụ lục A.1 của chính bài.} Phụ lục A.1,
    phương trình (4) và (5), cho \(h\) khoảng \([-1,1]\) và \(g\) khoảng
    \([-2,2]\). Chú thích bảng 10 thì ghi ngược lại, gọi \enquote{h1} là
    sigmoid trong \([-2,2]\) và \enquote{g2} là sigmoid trong \([-1,1]\). Sách
    theo phụ lục A.1, vì đó là chỗ in công thức chứ không phải chỗ in nhãn viết
    tắt.
  \item \textbf{Trạng thái ô nhớ không phải \(\vect{c}_t\) trong bài.} Bài viết
    \(s_{c_j^v}(t)\), với \(j\) đánh số khối và \(v\) đánh số cell trong khối.
    Ký hiệu \(c_j\) trong bài là \emph{tên của cell}, không phải giá trị trạng
    thái của nó. Sách viết \(\vect{c}_t\) cho trạng thái vì sách làm việc ở mức
    tầng chứ không mức từng cell.
\end{itemize}

\section{Hai chỗ dễ đọc nhầm ở Sutskever 2014}
\label{sec:app-a-sutskever}

\begin{itemize}
  \item \textbf{Chữ \(c\) đổi vai giữa hai bài đứng cạnh nhau.} Sutskever viết
    \(v\) cho vector ngữ cảnh, còn Bahdanau viết \(c\), và \(c\) trong sách này
    là trạng thái ô nhớ của chương~\ref{ch:lstm}. Đặt hai bài cạnh nhau mà giữ
    nguyên ký hiệu gốc thì \(c\) có ba nghĩa trên cùng một trang. Sách viết
    \(\vect{v}\) cho vector ngữ cảnh ở cả hai chương.
  \item \textbf{\enquote{The LSTM formulation from Graves} không phải bản 1997.}
    Bài chỉ dành một câu cho ô nhớ nó dùng, và câu đó trỏ sang Graves 2013, nơi
    phương trình (7) tới (11) có cổng quên của Gers 2000, có peephole của Gers
    2002, và dùng tanh ở cả hai chỗ nén. Đọc bài Sutskever với bản 1997 trong
    đầu là lệch mất ba thay đổi kiến trúc; xem mục~\ref{sec:ch04-cau-noi}.
\end{itemize}
