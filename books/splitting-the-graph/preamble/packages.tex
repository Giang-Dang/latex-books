% Code listings: minted v3 (Pygments-based). Needs -shell-escape and
% TEXMF_OUTPUT_DIRECTORY, both set in this book's .latexmkrc, and a Python
% with Pygments on the build machine.
\usepackage{minted}
\setminted{autogobble, breaklines, fontsize=\small, tabsize=2}

% SPEC decision 31. Pygments' graphql lexer handles executable documents
% (queries, mutations, fragments) and has no schema-definition-language
% support at all. Measured with Pygments 2.19.2 on 2026-08-19: a nine-line
% federation SDL snippet using @key, @external, extend type and "!" produces
% 77 Error tokens, rendered as red boxes. The same snippet through the ruby
% lexer produces 0: "#" comments, CamelCase type names, "@directive" names,
% quoted strings and block strings all land on sensible tokens.
%
% So SDL uses this environment, which borrows that lexer under an honest name.
% Executable GraphQL still uses minted's real graphql lexer. A custom Pygments
% lexer was rejected because minted v3 needs a per-machine .latexminted_config
% to load one, which would break a fresh clone and the pre-commit build gate.
% Revisit if Pygments gains an SDL lexer.
\newminted[graphqlsdl]{ruby}{}
